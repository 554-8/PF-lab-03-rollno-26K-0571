# pseudocodes   
### problem 1: student information using different data types 
start     
declare student name as string     
declare student age as integer     
declare student grade as char        
declare percentage as float          
set student name= "Zara"   
set student age = 18    
set student grade = 'A'    
set percentage = 86.5    
display student name     
display student age     
display student grade     
display percentage     
end
